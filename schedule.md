<script>
(function() {
  var root = document.getElementById('calendarCarousel');
  if (!root) return;
  var slides = Array.prototype.slice.call(root.querySelectorAll('.calendar-slide'));
  var titleEl = document.getElementById('calMonthTitle');
  var dotsWrap = document.getElementById('calDots');
  var current = 0;

  slides.forEach(function(slide, i) {
    var dot = document.createElement('button');
    dot.type = 'button';
    dot.className = 'calendar-dot';
    dot.setAttribute('aria-label', slide.getAttribute('data-title'));
    dot.addEventListener('click', function() { show(i); });
    dotsWrap.appendChild(dot);
  });

  function show(i) {
    current = (i + slides.length) % slides.length;
    slides.forEach(function(slide, idx) {
      slide.classList.toggle('is-active', idx === current);
    });
    titleEl.textContent = slides[current].getAttribute('data-title');
    Array.prototype.forEach.call(dotsWrap.children, function(dot, idx) {
      dot.classList.toggle('is-active', idx === current);
    });
  }

  document.getElementById('calPrevBtn').addEventListener('click', function() { show(current - 1); });
  document.getElementById('calNextBtn').addEventListener('click', function() { show(current + 1); });

  var now = new Date();
  var todayYm = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0');
  var startIdx = 0;
  slides.forEach(function(slide, i) {
    if (slide.getAttribute('data-ym') === todayYm) startIdx = i;
  });
  show(startIdx);
})();
</script>
